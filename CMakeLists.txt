cmake_minimum_required(VERSION 3.18)
project(expo_quickjs LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(QUICKJS_SANITIZE "Address/undefined sanitizers" OFF)
if(QUICKJS_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
set(QJS vendor/quickjs-2025-09-13)
add_library(quickjs STATIC ${QJS}/quickjs.c ${QJS}/dtoa.c ${QJS}/libregexp.c ${QJS}/libunicode.c ${QJS}/cutils.c)
target_compile_definitions(quickjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2025-09-13")

if(NOT APPLE)
 target_link_libraries(quickjs PUBLIC m)
endif()
add_library(expo-quickjs STATIC cpp/Execution.cpp)
target_include_directories(expo-quickjs PUBLIC cpp)
target_link_libraries(expo-quickjs PUBLIC quickjs)
if(ANDROID)
 add_library(expo-quickjs-jni SHARED android/src/main/cpp/QuickJSJNI.cpp)
 target_link_libraries(expo-quickjs-jni PRIVATE expo-quickjs log)
else()
 find_package(Threads REQUIRED)
 target_link_libraries(expo-quickjs PUBLIC Threads::Threads)
 enable_testing()
 add_executable(execution_test native-tests/execution_test.cpp)
 target_link_libraries(execution_test PRIVATE expo-quickjs)
 add_executable(channel_runner native-tests/channel_runner.cpp)
 target_link_libraries(channel_runner PRIVATE expo-quickjs)
 add_test(NAME execution COMMAND execution_test)
 set_tests_properties(execution PROPERTIES TIMEOUT 20)
endif()
